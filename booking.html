// ====== SETTINGS ====== const STAFF_PIN = '2027'; // <-- change me const SHEET_NAME = 'TMS DB'; const TICKET_COUNT = 2000; // Columns: A TicketID | B Note | C Status | D CheckedInAt | E QR image // ====== ONE-TIME SETUP (run from the editor) ====== function setup() { const ss = SpreadsheetApp.getActive(); let sheet = ss.getSheetByName(SHEET_NAME) || ss.insertSheet(SHEET_NAME); sheet.clear(); sheet.getRange(1, 1, 1, 5).setValues([['TicketID', 'Note', 'Status', 'CheckedInAt', 'QR']]); // No 0/O/1/I so codes are easy to type from a printed ticket const alphabet = 'ABCDEFGHJKLMNPQRSTUVWXYZ23456789'; const used = {}; const rows = []; while (rows.length < TICKET_COUNT) { const id = makeId(); if (used[id]) continue; used[id] = true; rows.push([id, '', 'UNUSED', '']); } sheet.getRange(2, 1, rows.length, 4).setValues(rows); sheet.setFrozenRows(1); CacheService.getScriptCache().remove('idmap'); SpreadsheetApp.flush(); } // Run after setup(). Adds a QR picture next to each ID (QR contains ONLY the ticket ID). function addQrCodes() { const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME); const n = sheet.getLastRow() - 1; const formulas = []; for (let r = 2; r < n + 2; r++) { formulas.push(['=IMAGE("https://api.qrserver.com/v1/create-qr-code/?size=200x200&margin=10&data="&A' + r + ')']); } sheet.getRange(2, 5, n, 1).setFormulas(formulas); sheet.setColumnWidth(5, 120); sheet.setRowHeights(2, n, 120); } // TESTING ONLY: marks every ticket UNUSED again. Run before the event, never during it. function resetAllTickets() { const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME); const n = sheet.getLastRow() - 1; const blank = []; for (let i = 0; i < n; i++) blank.push(['UNUSED', '']); sheet.getRange(2, 3, n, 2).setValues(blank); } // ====== WEB APP ====== function doGet(e) { const p = e.parameter || {}; if (p.action === 'book') return json(handleBooking(p)); if (p.action === 'stats') return json(stats()); if (p.action === 'list') return json(listTickets()); const id = (p.t || '').trim().toUpperCase(); if (!id) return json({ result: 'invalid' }); const markUsed = p.action !== 'lookup'; // action=lookup checks without using the ticket const lock = LockService.getScriptLock(); lock.waitLock(15000); try { const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME); let row = getMap(false)[id]; if (!row) row = getMap(true)[id]; // refresh once in case tickets were added if (!row) return json({ result: 'invalid' }); const cells = sheet.getRange(row, 2, 1, 3).getValues()[0]; // Note, Status, CheckedInAt const note = cells[0], status = cells[1], when = cells[2]; if (status === 'USED') return json({ result: 'used', time: String(when), note: note }); if (markUsed) { const now = Utilities.formatDate(new Date(), Session.getScriptTimeZone(), 'dd MMM HH:mm:ss'); sheet.getRange(row, 3, 1, 2).setValues([['USED', now]]); } return json({ result: 'valid', note: note, marked: markUsed }); } finally { lock.releaseLock(); } } // ====== HELPERS ====== function getMap(forceRefresh) { const cache = CacheService.getScriptCache(); if (!forceRefresh) { const hit = cache.get('idmap'); if (hit) return JSON.parse(hit); } const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME); const ids = sheet.getRange(2, 1, sheet.getLastRow() - 1, 1).getValues(); const map = {}; ids.forEach((r, i) => { map[String(r[0]).toUpperCase()] = i + 2; }); try { cache.put('idmap', JSON.stringify(map), 21600); } catch (e) {} return map; } function stats() { const cache = CacheService.getScriptCache(); const hit = cache.get('stats'); if (hit) return JSON.parse(hit); const sheet = SpreadsheetApp.getActive().getSheetByName(SHEET_NAME); const status = sheet.getRange(2, 3, sheet.getLastRow() - 1, 1).getValues().flat(); const out = { result: 'stats', used: status.filter(s => s === 'USED').length, total: status.length }; cache.put('stats', JSON.stringify(out), 15); return out; } function json(obj) { return ContentService.createTextOutput(JSON.stringify(obj)) .setMimeType(ContentService.MimeType.JSON); }
